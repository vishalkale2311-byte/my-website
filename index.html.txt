    <script>

        /* --------------------------------
           DEFAULT BLOGS
        -------------------------------- */

        let blogs =
            JSON.parse(
                localStorage.getItem("blogs")
            ) || [

                {
                    title:
                        "Introduction to Web Development",

                    author:
                        "Rocky",

                    category:
                        "Technology",

                    content:
                        "Web development is the process of creating websites and web applications using HTML, CSS and JavaScript.",

                    likes: 0
                },

                {
                    title:
                        "Why Learn JavaScript?",

                    author:
                        "Admin",

                    category:
                        "Programming",

                    content:
                        "JavaScript is a popular programming language used to create interactive and dynamic websites.",

                    likes: 0
                },

                {
                    title:
                        "Importance of Education",

                    author:
                        "BlogHub",

                    category:
                        "Education",

                    content:
                        "Education helps people develop knowledge, skills and confidence to achieve their goals.",

                    likes: 0
                }

            ];


        /* --------------------------------
           SAVE BLOGS
        -------------------------------- */

        function saveBlogs() {

            localStorage.setItem(
                "blogs",
                JSON.stringify(blogs)
            );

        }


        /* --------------------------------
           DISPLAY BLOGS
        -------------------------------- */

        function displayBlogs(
            blogList = blogs
        ) {

            const container =
                document.getElementById(
                    "blogContainer"
                );

            container.innerHTML = "";


            if (blogList.length === 0) {

                container.innerHTML = `

                    <div class="no-blog">

                        <h3>
                            No blogs found
                        </h3>

                        <p>
                            Try another search.
                        </p>

                    </div>

                `;

                return;
            }


            blogList.forEach(
                (blog, index) => {

                    container.innerHTML += `

                        <article
                            class="blog-card">

                            <div
                                class="blog-image">

                                📝

                            </div>


                            <div
                                class="blog-content">

                                <span
                                    class="blog-category">

                                    ${escapeHTML(
                                        blog.category
                                    )}

                                </span>


                                <h3>

                                    ${escapeHTML(
                                        blog.title
                                    )}

                                </h3>


                                <div
                                    class="blog-author">

                                    By
                                    ${escapeHTML(
                                        blog.author
                                    )}

                                </div>


                                <p>

                                    ${escapeHTML(
                                        getShortContent(
                                            blog.content
                                        )
                                    )}

                                </p>


                                <div
                                    class="blog-actions">


                                    <button
                                        class="like-btn"
                                        onclick="likeBlog(${index})">

                                        ❤️
                                        ${blog.likes}

                                    </button>


                                    <button
                                        class="edit-btn"
                                        onclick="editBlog(${index})">

                                        ✏️ Edit

                                    </button>


                                    <button
                                        class="delete-btn"
                                        onclick="deleteBlog(${index})">

                                        🗑️ Delete

                                    </button>


                                </div>

                            </div>

                        </article>

                    `;

                }
            );

        }


        /* --------------------------------
           SHORT CONTENT
        -------------------------------- */

        function getShortContent(
            content
        ) {

            if (content.length > 120) {

                return (
                    content.substring(
                        0,
                        120
                    ) + "..."
                );

            }

            return content;

        }


        /* --------------------------------
           ADD / UPDATE BLOG
        -------------------------------- */

        document
            .getElementById("blogForm")
            .addEventListener(
                "submit",
                function(event) {

                    event.preventDefault();


                    const title =
                        document
                            .getElementById("title")
                            .value
                            .trim();


                    const author =
                        document
                            .getElementById("author")
                            .value
                            .trim();


                    const category =
                        document
                            .getElementById("category")
                            .value;


                    const content =
                        document
                            .getElementById("content")
                            .value
                            .trim();


                    const editIndex =
                        document
                            .getElementById(
                                "editIndex"
                            )
                            .value;


                    /* UPDATE */

                    if (editIndex !== "") {

                        blogs[editIndex].title =
                            title;

                        blogs[editIndex].author =
                            author;

                        blogs[editIndex].category =
                            category;

                        blogs[editIndex].content =
                            content;


                        alert(
                            "Blog updated successfully!"
                        );

                    }

                    /* ADD */

                    else {

                        const newBlog = {

                            title: title,

                            author: author,

                            category: category,

                            content: content,

                            likes: 0

                        };


                        blogs.unshift(
                            newBlog
                        );


                        alert(
                            "Blog published successfully!"
                        );

                    }


                    saveBlogs();

                    displayBlogs();

                    resetForm();


                    document
                        .getElementById(
                            "blogs"
                        )
                        .scrollIntoView({
                            behavior: "smooth"
                        });

                }
            );


        /* --------------------------------
           DELETE BLOG
        -------------------------------- */

        function deleteBlog(index) {

            const result =
                confirm(
                    "Are you sure you want to delete this blog?"
                );


            if (result) {

                blogs.splice(
                    index,
                    1
                );

                saveBlogs();

                displayBlogs();

                alert(
                    "Blog deleted successfully!"
                );

            }

        }


        /* --------------------------------
           LIKE BLOG
        -------------------------------- */

        function likeBlog(index) {

            blogs[index].likes++;

            saveBlogs();

            displayBlogs();

        }


        /* --------------------------------
           EDIT BLOG
        -------------------------------- */

        function editBlog(index) {

            const blog =
                blogs[index];


            document
                .getElementById("title")
                .value =
                blog.title;


            document
                .getElementById("author")
                .value =
                blog.author;


            document
                .getElementById("category")
                .value =
                blog.category;


            document
                .getElementById("content")
                .value =
                blog.content;


            document
                .getElementById("editIndex")
                .value =
                index;


            document
                .getElementById(
                    "submitButton"
                )
                .textContent =
                "Update Blog";


            document
                .getElementById(
                    "create"
                )
                .scrollIntoView({
                    behavior: "smooth"
                });

        }


        /* --------------------------------
           RESET FORM
        -------------------------------- */

        function resetForm() {

            document
                .getElementById(
                    "blogForm"
                )
                .reset();


            document
                .getElementById(
                    "editIndex"
                )
                .value = "";


            document
                .getElementById(
                    "submitButton"
                )
                .textContent =
                "Publish Blog";

        }


        /* --------------------------------
           SEARCH
        -------------------------------- */

        function searchBlogs() {

            const searchText =
                document
                    .getElementById(
                        "searchInput"
                    )
                    .value
                    .toLowerCase()
                    .trim();


            const filteredBlogs =
                blogs.filter(
                    blog =>

                    blog.title
                        .toLowerCase()
                        .includes(
                            searchText
                        )

                    ||

                    blog.author
                        .toLowerCase()
                        .includes(
                            searchText
                        )

                    ||

                    blog.category
                        .toLowerCase()
                        .includes(
                            searchText
                        )

                    ||

                    blog.content
                        .toLowerCase()
                        .includes(
                            searchText
                        )
                );


            displayBlogs(
                filteredBlogs
            );

        }


        /* --------------------------------
           MOBILE MENU
        -------------------------------- */

        function toggleMenu() {

            document
                .querySelector(
                    ".nav-links"
                )
                .classList
                .toggle("active");

        }


        /* --------------------------------
           SECURITY
        -------------------------------- */

        function escapeHTML(text) {

            const div =
                document.createElement(
                    "div"
                );

            div.textContent = text;

            return div.innerHTML;

        }


        /* --------------------------------
           START WEBSITE
        -------------------------------- */

        displayBlogs();

    </script>
